\section{Improvement 2: a hybrid metric--parametric trial space}
\label{sec:hybrid}

\subsection{Requirements and degrees of freedom in the choice of the space}

A metric trial space $\mathcal U_e$ for the formulation of \cref{sec:formulation}
must satisfy
\begin{enumerate}
\item[(R1)] $\dim\mathcal U_e=n=(p_\xi+1)(p_\eta+1)$ and unisolvence at the element
  nodes (interpolatory shape functions, shared nodal unknowns, diagonal GLL mass);
\item[(R2)] $\Pp_m(x,y)\subset\mathcal U_e$ with $m=\min(p_\xi,p_\eta)$ (completeness,
  \cref{prop:immunity}; contains the rigid-body modes);
\item[(R3)] invariance under rigid-body motions and scaling of the element
  (frame invariance).
\end{enumerate}
(R2) fixes $(m+1)(m+2)/2$ of the $n$ dimensions; for $p_\xi=p_\eta=p$, the
remaining $p(p+1)/2$ dimensions are free. $\Qp_p(s,t)$ fills them with Cartesian
polynomials of total degree $p+1,\dots,2p$. On a non-affine element the nodes do not
form a tensor grid in $(s,t)$, and interpolation of these high-degree Cartesian
polynomials on the perturbed grid is what makes the Lebesgue constant grow
(\cref{sec:num-lebesgue}). In the parametric coordinates, on the other hand, the nodes
form the exact GLL tensor grid on every element. This motivates writing only the
complete polynomials in metric coordinates and the rest in parametric coordinates.

\subsection{Definition}

\begin{definition}[Parametric complement and hybrid space]
Let $L_a$ be the Legendre polynomials and
\begin{equation}
  \Cp_p=\spn\{L_a(\xi)L_b(\eta):\ 0\le a\le p_\xi,\ 0\le b\le p_\eta,\ a+b>m\}.
  \label{eq:complement}
\end{equation}
The hybrid trial space of element $e$ is defined through its pull-back to
$\hat\Omega$,
\begin{equation}
  \hat\Tp_p=\bigl\{q\circ\vect F_e:\ q\in\Pp_m(x,y)\bigr\}+\Cp_p ,
  \qquad
  \Tp_p=\bigl\{\hat v\circ\vect F_e^{-1}:\ \hat v\in\hat\Tp_p\bigr\},
  \label{eq:hybrid}
\end{equation}
and its shape functions are the functions $\hat M_l\in\hat\Tp_p$ with
$\hat M_l(\bxi_k)=\delta_{lk}$.
\end{definition}

The complete polynomials are represented by $L_a(\tilde s)L_b(\tilde t)$, $a+b\le m$,
in the scaled skew coordinates \eqref{eq:bbox}. The definition through
$\hat\Omega$ requires only the evaluation of $\vect F_e$, not its inverse. Where
$\vect F_e$ is not injective (folded elements), only the pull-back $\hat\Tp_p$ is
defined---the physical space $\Tp_p$ and point evaluation at physical coordinates are
not---and all element quantities are computed in reference coordinates, as in
\eqref{eq:Ke}.

\begin{lemma}[Canonical complement]
\label{lem:complement}
$\Cp_p$ is the $L^2(\hat\Omega)$-orthogonal complement of $\Pp_m(\xi,\eta)$ in
$\Qp_{p_\xi,p_\eta}(\xi,\eta)$. In particular, $\hat\Tp_p$ is defined independently of
any basis.
\end{lemma}
\begin{proof}
Let $a+b>m$ and $i+j\le m$. Then $i<a$ or $j<b$, and the orthogonality of $L_a$ to
all polynomials of degree $<a$ gives
$\int_{\hat\Omega}L_a(\xi)L_b(\eta)\,\xi^i\eta^j\dd\xi\dd\eta
=\int L_a\xi^i\dd\xi\int L_b\eta^j\dd\eta=0$. The functions $L_aL_b$ are linearly
independent, and $\dim\Cp_p=n-\dim\Pp_m$.
\end{proof}

For a non-affine map, $\Pp_m(x,y)\circ\vect F_e\ne\Pp_m(\xi,\eta)$, so a different
complement of $\Pp_m(\xi,\eta)$ (e.g.\ $\spn\{\xi^a\eta^b: a+b>m\}$) would give a
different space $\hat\Tp_p$. \Cref{lem:complement} identifies \eqref{eq:complement} as
the canonical choice; this is why, unlike for $\Qp_p(s,t)$, the basis of the complement
is part of the definition.

\subsection{Properties}

\begin{proposition}[Properties of the hybrid space]
\label{prop:hybrid}
Let the hybrid Vandermonde matrix be non-singular. Then
\begin{enumerate}
\item[(i)] $\dim\Tp_p=n$, the sum in \eqref{eq:hybrid} is direct, and the shape
  functions exist and are unique;
\item[(ii)] $\Pp_m(x,y)\subset\Tp_p$; in particular \cref{prop:immunity} holds with
  $q=m$, and the rigid-body modes are contained;
\item[(iii)] $\Tp_p$ is invariant under affine transformations of the element
  (\cref{lem:affine});
\item[(iv)] if $\vect F_e$ is bilinear (straight-edged quadrilateral), then
  $\Tp_p=\Qp_{p_\xi,p_\eta}(\xi,\eta)\circ\vect F_e^{-1}$: the hybrid shape functions
  coincide with the parametric ones, and \PH\ is the classical, symmetric spectral
  element.
\end{enumerate}
\end{proposition}
\begin{proof}
(i) $\hat\Tp_p$ is spanned by $n$ functions whose values at the $n$ nodes form the
non-singular matrix $\matr V$; hence they are linearly independent and the nodal map is
bijective. (ii) and (iii) are immediate from \eqref{eq:hybrid}, \cref{lem:affine} and
the invariance of $\bxi$. (iv) For bilinear $\vect F_e$, $x,y\in\Qp_1(\xi,\eta)$, hence
$x^ay^b\circ\vect F_e\in\Qp_{a+b}\subset\Qp_{p_\xi,p_\eta}$ for $a+b\le m$; with
$\Cp_p\subset\Qp_{p_\xi,p_\eta}$ we get $\hat\Tp_p\subset\Qp_{p_\xi,p_\eta}$, and equality
follows from (i).
\end{proof}

Property (iv) means that the unsymmetric formulation becomes active only where it
is needed, on elements with curved edges. On straight-edged meshes, however strong
the angular distortion, \PH\ inherits the symmetry, positive definiteness and
spectral properties of the classical element; $\Qp_p(s,t)$ does not have this property.

\subsection{Representation of the interpolation operator and Lebesgue constant}

Let $\hat\Pi$ denote GLL interpolation on $\hat\Omega$ (onto
$\Qp=\Qp_{p_\xi,p_\eta}(\xi,\eta)$), $\pi$ the projector of $\Qp$ onto
$\Pp_m(\xi,\eta)$ along $\Cp_p$ (truncation of the Legendre expansion), and
$\mathcal P=\{q\circ\vect F_e: q\in\Pp_m(x,y)\}$. Define
\begin{equation}
  S=\pi\hat\Pi\big|_{\mathcal P}:\ \mathcal P\to\Pp_m(\xi,\eta),\qquad
  S_Q=\hat\Pi\big|_{\hat\Qp}:\ \hat\Qp\to\Qp,\quad
  \hat\Qp=\{q\circ\vect F_e: q\in\Qp_p(s,t)\},
  \label{eq:S}
\end{equation}
and denote by $\mathcal I_T$, $\mathcal I_Q$ the nodal interpolation operators of
$\hat\Tp_p$ and $\hat\Qp$ (all functions pulled back to $\hat\Omega$).

\begin{proposition}[Representation of the metric interpolants]
\label{prop:representation}
Assume that $\vect F_e$ has a point with $\det\matr J\ne0$, so that
$\dim\mathcal P=\dim\Pp_m$.
\begin{enumerate}
\item[(a)] $\hat\Tp_p$ is unisolvent at the GLL nodes if and only if $S$ is invertible,
  and then, for every $f\in C(\hat\Omega)$,
  \begin{equation}
    \mathcal I_Tf=\hat\Pi f+(I-\hat\Pi)\,S^{-1}\pi\,\hat\Pi f .
    \label{eq:repT}
  \end{equation}
\item[(b)] $\hat\Qp$ is unisolvent if and only if $S_Q$ is invertible, and then
  \begin{equation}
    \mathcal I_Qf=\hat\Pi f+(I-\hat\Pi)\,S_Q^{-1}\,\hat\Pi f .
    \label{eq:repQ}
  \end{equation}
\item[(c)] Consequently, with operator norms on $(\Qp,\|\cdot\|_{L^\infty(\hat\Omega)})$,
  \begin{equation}
    \Lam_T\le\Lam_N\bigl(1+\|(I-\hat\Pi)S^{-1}\pi\|\bigr),\qquad
    \Lam_Q\le\Lam_N\bigl(1+\|(I-\hat\Pi)S_Q^{-1}\|\bigr),
    \label{eq:lebbound}
  \end{equation}
  and the shape functions are
  $\hat M_k^T=\hat N_k+(I-\hat\Pi)S^{-1}\pi\hat N_k$,
  $\hat M_k^Q=\hat N_k+(I-\hat\Pi)S_Q^{-1}\hat N_k$.
\end{enumerate}
\end{proposition}
\begin{proof}
The nodal values of a function $g$ are in one-to-one correspondence with
$\hat\Pi g\in\Qp$, and $\hat\Pi c=c$ for $c\in\Cp_p\subset\Qp$.
(a) Let $g=q+c$ with $q\in\mathcal P$, $c\in\Cp_p$, have the nodal values of $f$, i.e.\
$\hat\Pi q+c=\hat\Pi f$. Applying $\pi$ (whose kernel is $\Cp_p$) gives
$Sq=\pi\hat\Pi f$; conversely, for any $q$ with $Sq=\pi\hat\Pi f$ the function
$c=\hat\Pi f-\hat\Pi q$ lies in $\ker\pi=\Cp_p$. If $S$ is invertible, $q$ and $c$ are
thus uniquely determined for every $f$ (existence and uniqueness of the interpolant), and
$g=q+\hat\Pi f-\hat\Pi q$, which is \eqref{eq:repT}. If $S$ is singular, take
$0\ne q\in\ker S$ and $c=-\hat\Pi q\in\Cp_p$: the function $g=q+c$ vanishes at all nodes;
either $g\ne0$, or $q=-c\in\mathcal P\cap\Cp_p$ and the sum is not direct, so that
$\dim\hat\Tp_p<n$; in both cases $\hat\Tp_p$ is not unisolvent.
(b) An interpolant $g\in\hat\Qp$ satisfies $\hat\Pi g=\hat\Pi f$, i.e.\ $S_Qg=\hat\Pi f$,
and $g=\hat\Pi g+(I-\hat\Pi)g$. (c) follows from $\|\hat\Pi f\|\le\Lam_N\|f\|$ and,
for the shape functions, from $\hat\Pi f=\hat N_k$ for the nodal data $\delta_{\cdot k}$.
\end{proof}

\begin{corollary}
\label{cor:defect}
If $\mathcal P\subset\Qp$---in particular for every bilinear map---the correction in
\eqref{eq:repT} vanishes, $\mathcal I_T=\hat\Pi$ and $\Lam_T=\Lam_N$, independently of
the amount of (angular) distortion. In general, the deviation of $\mathcal I_T$ from
GLL interpolation is the GLL interpolation error of a complete polynomial of degree $m$,
$(I-\hat\Pi)(q\circ\vect F_e)$, $q\in\Pp_m(x,y)$, i.e.\ precisely the
\emph{completeness defect} of the classical element.
\end{corollary}

\Cref{prop:representation} puts both metric spaces in the same framework---GLL
interpolation plus a correction---and isolates the structural difference. For
$\Tp_p$, the correction has rank at most $\dim\Pp_m=(m+1)(m+2)/2$ and involves the
inverse $S^{-1}$ of the low-order block only; the high-order part of the interpolant is
untouched. For $\Qp_p(s,t)$, the correction acts on the whole $n$-dimensional space,
involves the GLL interpolation error of Cartesian polynomials of tensor degree $p$
(total degree up to $2p$) and the inverse of the full matrix $S_Q$. The bounds
\eqref{eq:lebbound} are not sharp enough to quantify the difference a priori; the
numerical results of \cref{sec:num-lebesgue} show that $\Lam_T$ stays at (or very
close to) $\Lam_N$ for moderate curved-edge distortion and grows for strong distortion,
but by orders of magnitude less than $\Lam_Q$, which grows exponentially in $p$ and in
the distortion.

\subsection{Approximation}

\begin{lemma}[Interpolation error]
\label{lem:interp}
Let $\mathcal U_e\in\{\Tp_p,\Qp_p(s,t)\}$ be unisolvent, $q_e$ its completeness
degree ($m$ resp.\ $p$) and $\Lam_{\mathcal U}$ its Lebesgue constant, and let
$\tilde\Omega_e$ be the convex hull of $\Omega_e$. Then for
$u\in W^{q_e+1,\infty}(\tilde\Omega_e)$
\begin{equation}
  \|u-\mathcal I_{\mathcal U}u\|_{L^\infty(\Omega_e)}
  \le(1+\Lam_{\mathcal U})\inf_{q\in\Pp_{q_e}(x,y)}\|u-q\|_{L^\infty(\Omega_e)}
  \le C\,(1+\Lam_{\mathcal U})\,h_e^{q_e+1}|u|_{W^{q_e+1,\infty}(\tilde\Omega_e)} .
  \label{eq:interp}
\end{equation}
Moreover, with the derivative Lebesgue constant
$\Lam'_{\mathcal U}=h_e\max_{\bx\in\Omega_e}\sum_l|\nabla M_l^e(\bx)|$,
\begin{equation}
  |u-\mathcal I_{\mathcal U}u|_{W^{1,\infty}(\Omega_e)}
  \le C\,(1+\Lam'_{\mathcal U})\,h_e^{q_e}|u|_{W^{q_e+1,\infty}(\tilde\Omega_e)} .
  \label{eq:interp1}
\end{equation}
For a family of elements obtained from a fixed element by similarity
transformations (self-similar refinement), $\Lam_{\mathcal U}$ and $\Lam'_{\mathcal U}$
are independent of $h_e$ by \cref{lem:affine}.
\end{lemma}
\begin{proof}
$\mathcal I_{\mathcal U}$ reproduces $\Pp_{q_e}(x,y)$, so
$u-\mathcal I_{\mathcal U}u=(u-q)-\mathcal I_{\mathcal U}(u-q)$ for every such $q$. Choose
$q$ as the Taylor polynomial of degree $q_e$ of $u$ about a point of $\tilde\Omega_e$; then
$\|u-q\|_{L^\infty}\le Ch_e^{q_e+1}|u|_{W^{q_e+1,\infty}}$ and
$|u-q|_{W^{1,\infty}}\le Ch_e^{q_e}|u|_{W^{q_e+1,\infty}}$ on $\tilde\Omega_e$, with $C$
depending only on $q_e$. The $L^\infty$ estimate follows from
$\|\mathcal I_{\mathcal U}(u-q)\|_\infty\le\Lam_{\mathcal U}\|u-q\|_\infty$, the
$W^{1,\infty}$ estimate from
$|\nabla\mathcal I_{\mathcal U}(u-q)|\le\|u-q\|_\infty\sum_l|\nabla M_l^e|
\le h_e^{-1}\Lam'_{\mathcal U}\|u-q\|_\infty$. Under a similarity
$\bx\mapsto\alpha\matr R\bx+\vect c$ the shape functions are unchanged as functions of the
material point (\cref{lem:affine}), their gradients scale with $\alpha^{-1}$ and $h_e$ with
$\alpha$, so $\Lam'_{\mathcal U}$ is invariant.
\end{proof}

Both metric spaces thus have the optimal interpolation order, in $L^\infty$ and in
$W^{1,\infty}$ (hence in the energy norm), on meshes whose distortion persists under
refinement, where the classical element is limited to
$\Pp_k(x,y)$ with $gk\le p$; they differ in the constants $1+\Lam_{\mathcal U}$ and $1+\Lam'_{\mathcal U}$. An
error estimate for the Petrov--Galerkin solution additionally requires a discrete
inf-sup condition for the pair (test space, non-conforming trial space) and a
consistency estimate for the non-conforming trial space (Strang-type arguments). We do
not prove such an estimate for either space; \cref{sec:num-convergence,sec:num-spectra}
assess stability numerically.

\subsection{Gradients, stiffness matrix and folded elements}
\label{sec:hybrid-gradients}

For $\phi=L_a(\tilde s)L_b(\tilde t)\in\Pp_m$,
$\nabla_{\bx}\phi=\matr J_0^{-1}\mathrm{diag}(r_s^{-1},r_t^{-1})\nabla_{\tilde{\vect s}}\phi$;
for $\phi\in\Cp_p$, $\nabla_{\bx}\phi=\matr J(\bxi)^{-1}\nabla_{\bxi}\phi$ at the
quadrature point $\bxi$. Since the hybrid functions are evaluated through $\bxi$,
the stiffness matrix \eqref{eq:Ke} requires no inverse mapping, and post-processing at
arbitrary physical points requires $\vect F_e^{-1}$ only as in the classical element.
The complement contributes rational integrands to $\matr K_e$ (as for curved
isoparametric elements). This does not affect \cref{prop:immunity}: for
$\vect u\in[\Pp_m(x,y)]^2$ the coefficients of the interpolant on $\Cp_p$ vanish
(direct sum), $\matr B_R\hat{\vect u}=\bm\varepsilon(\vect u)$, and the integrand
$\tilde{\matr B}_L^\T\matr C\bm\varepsilon(\vect u)$ is the same polynomial as for
$\Qp_p(s,t)$. The sign of $\det\matr J$ is irrelevant for the function values;
$\det\matr J=0$ at a quadrature point is excluded because the gradients of $\Cp_p$
involve $\matr J^{-1}$.
