\chapter{Code reference}
\label{ch:code}

All routines are in \code{code/subroutines}. \code{Start_Sim.m} adds this folder
with all subfolders to the path (\code{genpath}). Scripts (no \code{function}
header) operate on the base workspace.

\section{Element technology}

\begin{xltabular}{\linewidth}{@{}P{6.4cm}L@{}}
  \toprule
  routine & purpose \\
  \midrule
  \endhead
  \code{ElementMatrices/Ke_2D(p, C, x, y [, nIPxi, nIPeta])} & unsymmetric element
    stiffness $\int \tilde{\matr B}_L^{\T}\matr C\matr B_R$ (default $2p$ Gauss
    points per direction) \\
  \code{ElementMatrices/Me_2D(p, rho, x, y, lumping [, nIPxi, nIPeta])} &
    unsymmetric element mass, lumping 1--4 \\
  \code{ElementMatrices/KMe_2D(p, C, rho, x, y, lumping)} & stiffness and mass in
    one loop (dynamic, modal and frequency-domain analyses) \\
  \code{ShapeFunctions/2D/ShapeFunctions2D_SEM(xi, eta, p)} & parametric (test)
    shape functions and $\partial/\partial\xi$, $\partial/\partial\eta$;
    \code{[nNodes x nPoints]} arrays \\
  \code{ShapeFunctions/2D/ShapeFunctions2D_Metric(xQ, yQ, xN, yN, p [, method, xiQ])} &
    metric (trial) shape functions and $\partial/\partial x$, $\partial/\partial y$;
    \code{method} 1 = $Q_p$ in skew coordinates (default), 0 = global coordinates,
    2 = hybrid space (\cref{sec:metric-hybrid}; \code{xiQ}: local coordinates of the
    query points); empty \code{method}: \code{p.metricMethod}; basis of the
    Vandermonde matrix from \code{p.metricBasis} (default \code{'lagrange'},
    \cref{ch:metric-basis}); optional 4th output $\matr V$ \\
  \code{ShapeFunctions/2D/ReferenceNodes2D(p)} & local coordinates of the element
    nodes in the order of \code{ShapeFunctions2D_SEM} \\
  \code{ShapeFunctions/2D/CorotationalSkewCoordinates(xQ, yQ, xN, yN, p)} & skew
    coordinates $(s,t)$ and $\matr J_0$ \\
  \code{ShapeFunctions/2D/ShapeFunctions2D_4nodes}, \code{_8nodes} & bilinear and
    serendipity functions (geometry of the input mesh) \\
  \code{ShapeFunctions/2D/ShapeFunctions2D_Metric_Tria} & unfinished metric
    functions for triangles ($p\le2$, not used) \\
  \code{ShapeFunctions/1D/LagrangePolynomials(x)} & 1D Lagrange polynomials
    (barycentric form) as function handles \\
  \code{ShapeFunctions/1D/ShapeFunctions1D_SEM(p, xi, ndist)} & 1D shape
    functions, order \code{[xi=-1, xi=+1, interior]} \\
  \code{ShapeFunctions/1D/PolynomialBasis1D(s, n, basis, ndist)} & 1D bases
    (monomial, Legendre, Chebyshev, Lagrange on the reference nodes) and their
    derivatives, vectorized \\
  \code{GeometryMapping/Global2LocalNewton(xQ, yQ, xN, yN, p)} & inverse mapping
    $\vect x\mapsto(\xi,\eta)$ by Newton's method, vectorized, no toolbox \\
  \code{GeometryMapping/Jacobi_Matrix(p, [xi eta], x, y)} &
    $\matr J = [x_{,\xi}\ y_{,\xi};\ x_{,\eta}\ y_{,\eta}]$ \\
  \code{GeometryMapping/Local2GlobalCoordinatesQuad(p, P, [x y])} &
    $\vect x(\xi,\eta)$ \\
  \code{GeometryMapping/Global2LocalCoordinatesQuad} & inverse map with
    \code{fsolve} (Optimization Toolbox) \\
  \code{Quadrature/lgwt(n, a, b)} & Gauss--Legendre points and weights
    (G.~von Winckel) \\
  \code{Quadrature/lglnodes(N)} & $N+1$ Gauss--Lobatto--Legendre points and
    weights (G.~von Winckel) \\
  \code{NodalDistribution/nodal_distribution(p, ndist)} & supporting points on
    $[-1,1]$ \\
  \code{Lumping/RowSum_Method}, \code{HRZ_Method} & lumping of an element mass
    matrix \\
  \bottomrule
\end{xltabular}

\section{Mesh and topology}

\begin{xltabular}{\linewidth}{@{}P{6.4cm}L@{}}
  \toprule
  routine & purpose \\
  \midrule
  \endhead
  \code{Topology/pSpectralElement_NodesTopology(T, x, y, p, p_org)} & creates the
    edge and interior nodes of degree $p$ from the 4-node ($p_\text{org}=1$) or
    8-node ($p_\text{org}=2$) mesh; shared edges get the same nodes, in flipped
    order for opposite orientation \\
  \code{Topology/Topology_Edge_2D(T)} & global edge numbers \code{[nElem x 4]}
    (edges 1-2, 2-3, 3-4, 4-1) \\
  \code{Topology/ConnectivityEdgeElement(TE)} & elements and local edge numbers of
    every edge \\
  \code{Topology/ConnectivityNodeElement(T)} & elements of every node \\
  \code{Topology/Boundary_Edges(TE)} & boundary edges
    \code{[element, local edge, edge]} \\
  \code{Topology/Topology_DoF_2D(T)} & DOF connectivity, node $n\mapsto$ DOFs
    $2n-1$, $2n$ \\
  \code{MeshDistortion/distort_mesh(X, T, type, ...)} & random distortion of
    interior nodes with a $\det\matr J$ safeguard (\code{type} \code{'Q4'} or
    \code{'Q8'}; stand-alone) \\
  \code{MeshDistortion/distort_single_element} & distortion of the element
    closest to the centre \\
  \bottomrule
\end{xltabular}

\section{Assembly, boundary conditions and loads}

\begin{xltabular}{\linewidth}{@{}P{6.4cm}L@{}}
  \toprule
  routine & purpose \\
  \midrule
  \endhead
  \code{Assembly/AssemblyFEM(T, p, x, y, nElem, nDoF, C, rho, nMat, mass_matrix, elsets, alpha, beta, zeta, flag_comp)} &
    sparse global $\matr K$, $\matr M$, $\matr R = \alpha\matr M + \beta\matr K$
    (only elements with damping). \code{flag_comp}: 0 = $\matr K$; 1 = $\matr K$,
    $\matr M$; 2 = $\matr K$, $\matr M$ and the critical time step
    $\Delta t = 0.99\cdot 2/\omega_{\max}$ from element eigenvalues. Optional
    6th output \code{ElemData}: element matrices and DOF connectivity (adaptive
    hybrid explicit method) \\
  \code{BoundaryConditions/Dirichlet/DirichletBCs_homogeneous(bc)} & constrained
    DOFs \code{[DOF_C, NOI]} (\code{fixed}, \code{UxUy 0}, \code{XSYMM}/\code{Ux 0},
    \code{YSYMM}/\code{Uy 0}; \code{noi} gives the nodes of interest) \\
  \code{BoundaryConditions/Dirichlet/DirichletBCs_nonhomogeneous(K, F, nDoF, bc, x, y)} &
    prescribed values by elimination: $\vect F \leftarrow \vect F - \matr
    K_{:,b}\bar{\vect u}_b$, rows and columns $b$ set to zero, diagonal kept,
    $F_b = K_{bb}\bar u_b$ \\
  \code{BoundaryConditions/Dirichlet/DirichletBCs_nonhomogeneous_dynamic(K, C, M, ...)} &
    lifting of constant and time-dependent prescribed displacements; load parts
    multiplied by $s(t)$, $\dot s(t)$, $\ddot s(t)$ \\
  \code{BoundaryConditions/Dirichlet/SpatialDisplacementFunction} & polynomial
    equilibrium fields for higher-order patch tests ($p$ and $\nu = 0.3$
    hard-coded) \\
  \code{BoundaryConditions/Dirichlet/HigherOrderDisplacementField.m} & script
    (Symbolic Math Toolbox) that generates those fields \\
  \code{BoundaryConditions/Neumann/NeumannBCs(p, cload, dsload, body_force, gravity, MAT, nDoF, T, x, y, nNodes [, flag_mRHS])} &
    load vectors \code{[F, F_dt]} (constant and time-dependent part); with
    \code{flag_mRHS} one column per load \\
  \code{CreateForce}, \code{CreatePress}, \code{CreateTraction},
    \code{CreateBload}, \code{CreateGravity} & individual loads (edge integrals
    with $p+1$ Gauss points, area integrals with $(p+1)^2$) \\
  \code{BoundaryConditions/Neumann/SpatialLoadingFunction} & spatially varying
    edge loads (Lee \& Bathe cubic patch test) \\
  \code{BoundaryConditions/pSpectralElement_BCs(dsload, bc, T, p, p_org)} & adds
    the high-order edge nodes to the node sets of loads and boundary conditions
    that have an element set \\
  \bottomrule
\end{xltabular}

The analyses and time integrators are described in \cref{ch:analyses}.

\section{Post-processing and utilities}

\begin{xltabular}{\linewidth}{@{}P{6.4cm}L@{}}
  \toprule
  routine & purpose \\
  \midrule
  \code{Visualization/Contour_plot(p, x, y, U, scale, fields, T, C, elsets, flag_ave, alpha)} &
    contour plots on the deformed mesh (\code{'ux'}, \code{'uy'}, \code{'mag'},
    \code{'Mises'}, \code{'sigma_x'}, \code{'sigma_y'}, \code{'tau_xy'},
    \code{'epsilon_x'}, \code{'epsilon_y'}, \code{'gamma_xy'}), evaluated with the

    boundary conditions \\
  \code{Visualization/Colormap} & diverging colour map (Moreland),
    \code{maxdistcolor} (S.~Cobeldick, BSD licence) \\
  \code{Misc/strain_energy(K, u)} & $\tfrac12\vect u^{\T}\matr K\vect u$ \\
  \code{Misc/RotateFeMatrix}, \code{Misc/quadmesh} & utilities \\
  \code{MMS_2D/} & method of manufactured solutions (Symbolic Math Toolbox):
    displacement and body-load functions for convergence studies; not connected
    to \code{Start_Sim} \\

\end{xltabular}
